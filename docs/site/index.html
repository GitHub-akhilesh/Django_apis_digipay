🌐 Web Components

Universal `` element supporting Floating, Sidebar, and Inline modes.

1. Web Component (1-Line Integration)

<!-- Load SDK & Component Scripts -->
<script src="digipay-chat-sdk.js"></script>
<script src="digipay-chat-widget.js"></script>

<!-- Embed Floating Chat Widget -->
<digipay-chat csc-id="500100100014" mode="floating"></digipay-chat>

2. React Integration Example

import React from 'react';
import { DigiPayChatWidget } from '@digipay/chat-react';

export default function Dashboard() {
  return (
    <div>
      <h1>Merchant Dashboard</h1>
      <DigiPayChatWidget cscId="500100100014" mode="floating" theme="dark" />
    </div>
  );
}
